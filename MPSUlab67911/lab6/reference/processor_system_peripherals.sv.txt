module processor_system(
  input  logic        clk_i,
  input  logic        resetn_i,

  // Входы и выходы периферии
  input  logic [15:0] sw_i,       // Переключатели

  output logic [15:0] led_o,      // Светодиоды

  input  logic        kclk_i,     // Тактирующий сигнал клавиатуры
  input  logic        kdata_i,    // Сигнал данных клавиатуры

  output logic [ 6:0] hex_led_o,  // Вывод семисегментных индикаторов
  output logic [ 7:0] hex_sel_o,  // Селектор семисегментных индикаторов

  input  logic        rx_i,       // Линия приёма по UART
  output logic        tx_o,       // Линия передачи по UART

  output logic [3:0]  vga_r_o,    // Красный канал vga
  output logic [3:0]  vga_g_o,    // Зелёный канал vga
  output logic [3:0]  vga_b_o,    // Синий канал vga
  output logic        vga_hs_o,   // Линия горизонтальной синхронизации vga
  output logic        vga_vs_o    // Линия вертикальной синхронизации vga
  
  
);

import peripheral_pkg::*;

logic sysclk, rst;
sys_clk_rst_gen divider(
    .ex_clk_i(clk_i),
    .ex_areset_n_i(resetn_i),
    .div_i(5),
    .sys_clk_o(sysclk),
    .sys_reset_o(rst)
);

logic irq_req;
logic irq_ret;

logic mem_req;
logic core_req;

logic mem_we;
logic core_we;

logic [31:0] mem_addr;
logic [31:0] core_addr;

logic [31:0] mem_wd;
logic [31:0] core_wd;


logic [31:0] mem_rd;
logic [31:0] ps2_rd;
logic [31:0] led_rd;

logic [31:0] dev_rd;
logic [31:0] core_rd;

logic [3:0] mem_be;
logic [2:0] core_size;


logic [31:0] instr;
logic [31:0] instr_addr;

logic stall;
logic ready;

logic [31:0] shrt_addr;
assign shrt_addr = {8'd0, mem_addr[23:0]};

logic [255:0] one_hot;
assign one_hot = 256'd1 << mem_addr[31:24];

instr_mem instr_mem(
    .read_addr_i(instr_addr),
    .read_data_o(instr)
);

processor_core core(
  .clk_i(sysclk),
  .rst_i(rst),
  
  .stall_i(stall),
  .irq_req_i(irq_req),
  .instr_i(instr),
  .mem_rd_i(core_rd),

  .irq_ret_o(irq_ret),
  .instr_addr_o(instr_addr),
  .mem_addr_o(core_addr),
  .mem_size_o(core_size),
  .mem_req_o(core_req),
  .mem_we_o(core_we),
  .mem_wd_o(core_wd)
);

data_mem data_mem(
  .clk_i(sysclk),
  .mem_req_i(mem_req & one_hot[0]),
  .write_enable_i(mem_we),
  .byte_enable_i(mem_be),
  .addr_i(shrt_addr),
  .write_data_i(mem_wd),
  .read_data_o(mem_rd),
  .ready_o()
);

lsu lsu(
  .clk_i(sysclk),
  .rst_i(rst),
  .core_req_i(core_req),
  .core_we_i(core_we),
  .core_size_i(core_size),
  .core_addr_i(core_addr),
  .core_wd_i(core_wd),
  .core_rd_o(core_rd),
  .core_stall_o(stall),
  .mem_req_o(mem_req),
  .mem_we_o(mem_we),
  .mem_be_o(mem_be),
  .mem_addr_o(mem_addr),
  .mem_wd_o(mem_wd),
  .mem_rd_i(dev_rd),
  .mem_ready_i(1'b1)
);

ps2_sb_ctrl ps2_sb_ctrl(
  .clk_i(sysclk),
  .rst_i(rst),
  .req_i(mem_req & one_hot[3]),
  .write_enable_i(mem_we),
  .addr_i(shrt_addr),
  .write_data_i(mem_wd),
  .read_data_o(ps2_rd),
  
  .interrupt_request_o(irq_req),
  .interrupt_return_i(irq_ret),

  .kclk_i(kclk_i),
  .kdata_i(kdata_i)
);

led_sb_ctrl led_sb_ctrl(
  .clk_i(sysclk),
  .rst_i(rst),
  .req_i(mem_req & one_hot[2]),
  .write_enable_i(mem_we),
  .addr_i(shrt_addr),
  .write_data_i(mem_wd),
  .read_data_o(led_rd),
  
  .led_o(led_o)
);

always_comb begin
    case(mem_addr[31:24])
        DMEM_ADDR_HIGH: dev_rd = mem_rd;
        LED_ADDR_HIGH: dev_rd = led_rd;
        PS2_ADDR_HIGH: dev_rd = ps2_rd;
        default: dev_rd = 32'd0;
    endcase
end
endmodule
